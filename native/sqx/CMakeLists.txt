cmake_minimum_required(VERSION 3.24)
project(SqxBridge LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "SQX bridge requires Windows")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(SqxBridge SHARED SqxBridge.cpp)
target_compile_definitions(SqxBridge PRIVATE UNICODE _UNICODE)
target_link_libraries(SqxBridge PRIVATE shell32)
