cmake_minimum_required(VERSION 3.24)
project(ZipGeniusSfx LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set_source_files_properties(stub.rc PROPERTIES OBJECT_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/default.ico;${CMAKE_CURRENT_SOURCE_DIR}/third_party/7z.exe;${CMAKE_CURRENT_SOURCE_DIR}/third_party/7z.dll")
add_executable(sfx_stub WIN32 stub.cpp stub.rc)
target_link_libraries(sfx_stub PRIVATE shell32 comctl32 ole32 uxtheme dwmapi gdiplus)
target_compile_definitions(sfx_stub PRIVATE UNICODE _UNICODE)
include(CTest)
if(BUILD_TESTING)
  add_executable(sfx_path_tests path_tests.cpp)
  target_compile_definitions(sfx_path_tests PRIVATE UNICODE _UNICODE)
  add_test(NAME sfx_paths COMMAND sfx_path_tests)
endif()
